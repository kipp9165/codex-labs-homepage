def run_codex_commerce_v1(raw_actions, founder_gate):
    paid_actions = [action for action in raw_actions if bool(action.get("paid", False))]
    metered_units = sum(int(action.get("units", 1)) for action in paid_actions)
    token_seed = "|".join(str(action.get("id", "")) for action in sorted(paid_actions, key=lambda a: str(a.get("id", ""))))
    session_token = "tok_" + str(len(token_seed))
    return {
        "version": "v1.0",
        "founder_gate": founder_gate,
        "paid_interactions": len(paid_actions),
        "metered_units": metered_units,
        "session_token": session_token,
    }

def build_product_registry(products, stripe_records):
    entries = []
    for product, stripe in zip(products, stripe_records):
        entries.append(
            {
                "product_id": product.get("product_id"),
                "family": product.get("family"),
                "edition": product.get("edition"),
                "stripe_product_code": stripe.get("stripe_product_code"),
                "mint_status": stripe.get("mint_status"),
            }
        )
    return {
        "registry_version": "v1.0",
        "count": len(entries),
        "entries": entries,
    }

===FILE=== codex_os\v1_0\commerce\founder_override_layer.py
"""Founder Override Layer."""


def apply_founder_override(policy_context, governance_state):
    sensitive_domains = {
        "business",
        "deals",
        "partnerships",
        "collaborations",
        "money",
        "strategic",
    }
    requested_domains = set(policy_context.get("requested_domains", []))
    requires_override = bool(requested_domains.intersection(sensitive_domains))
    founder_approved = bool(policy_context.get("founder_approved", False))
    granted = not requires_override or founder_approved
    return {
        "requires_founder_override": requires_override,
        "founder_approved": founder_approved,
        "override_granted": granted and governance_state.get("disposition") in {"APPROVE", "REVIEW"},
    }
===END===

def run_stripe_mint_remint(product_payload, founder_gate):
    product_id = product_payload.get("product_id", "CPG-CORE-1")
    can_issue = bool(founder_gate.get("override_granted", False)) and bool(product_payload.get("mintable", False))

    return {
        "stripe_product_code": stripe_product_code,
        "mint_status": "MINTED" if can_issue else "BLOCKED",
        "remint_status": "READY" if can_issue else "LOCKED",

===FILE=== codex_os\v1_0\crisis_simulation\crisis_simulation.py
"""Crisis Simulation."""


def run_crisis_simulation(simulation_state, governance_state):
    unstable_layers = [
        name
        for name, payload in simulation_state.items()
        if payload.get("state", {}).get("stability_status") != "STABLE"
    ]
    severity = "HIGH" if unstable_layers and governance_state.get("disposition") != "APPROVE" else "MODERATE"
    return {
        "unstable_layers": unstable_layers,
        "severity": severity,
        "disposition": "CRISIS_ESCALATE" if severity == "HIGH" else "CRISIS_MONITOR",
    }
===END===

"""Codex Provenance Chain."""


def build_provenance_chain(raw_actions, authenticity_state):
    action_ids = [str(action.get("id", "")) for action in sorted(raw_actions, key=lambda item: str(item.get("id", "")))]
    chain = []
    prior = "root"
    for action_id in action_ids:
        current = prior + "->" + action_id
        chain.append({"from": prior, "to": action_id, "edge": current})
        prior = current
    return {
        "authenticity": authenticity_state.get("authenticity_disposition", "INAUTHENTIC"),
        "chain": chain,
    }
===END===
